Criação de Componentes

Componentes são blocos reutilizáveis que constroem a interface do seu app. Em React Native, usamos componentes nativos como <View> e <Text> para estruturar a UI. 

// HelloWorld.js
import React from 'react';
import { View, Text } from 'react-native';
    
export default function HelloWorld(){
    return (
        <View>
            <Text>Olá, Mundo!</Text>
        </View>
    );
}
Arquivo principal app.js
            // App.js
import React from 'react';
import { SafeAreaView } from 'react-native';
import HelloWorld from './HelloWorld';

export default function App() {
    return (
        <SafeAreaView>
            <HelloWorld />
        </SafeAreaView>
    );
}

  • Exporte o componente para uso em outros arquivos.

import React from 'react';
import { View, Text } from 'react-native';

const Card = () => (

Uso de Props

Props permitem a passagem de dados para os componentes, tornando-os dinâmicos e personalizáveis.

  • Personalização e reutilização
  • Passagem de informações via atributos
import React from 'react';
import { SafeAreaView } from 'react-native';
import Card from './Card';

export default function App() {
    return (
        <SafeAreaView>
            <Card title="Meu Cartão" description="Esta é a descrição do cartão."/>
        </SafeAreaView>
    );
}



//- ----------------------------------------------------------------------------
// Card.js
import React from 'react';
import { View, Text } from 'react-native';

// Exemplo de componente com props
const Card = ({ title, description }) => (
    <View style={{ padding: 10, borderWidth: 1, borderColor: '#fff' }}>
        <Text>{title}</Text>
        <Text>{description}</Text>
    </View>
);
