Aula 15 / Context API

O que é Context API?

  • Forma de compartilhar dados entre componentes
  • Evita "prop drilling" (passar props em vários níveis)
  • Ideal para dados globais: temas, usuário logado, idioma

Crie um aplicativo com:

  • Um ThemeContext para gerenciar tema claro/escuro

Solução: Sistema de Tema

ThemeContext.jsCódigo
// ThemeContext.js
import { createContext } from 'react';
export const ThemeContext = createContext();

// App.js
const App = () => {
  const [theme, setTheme] = useState('light');
  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
  <Header />
  <Content />
    </ThemeContext.Provider>
  );
};

// Header.js
const Header = () => {
  const { theme, setTheme } = useContext(ThemeContext);
  return (
    <View style={{
  backgroundColor: theme === 'light' ? '#fff' : '#333',
  padding: 20
    }}>
  <Button
    title={theme === 'light' ? 'Tema escuro' : 'Tema claro'}
    onPress={() => setTheme(theme === 'light' ? 'dark' : 'light')}
  />
    </View>
  );
};

// Content.js
const Content = () => {
  const { theme } = useContext(ThemeContext);
  return (
    <View style={{
  flex: 1,
  backgroundColor: theme === 'light' ? '#f0f0f0' : '#222',
  padding: 20
    }}>
  <Text style={{ color: theme === 'light' ? '#000' : '#fff' }}>
    Conteúdo do App
  </Text>
    </View>
  );
};

Solução: Contador Global

CounterContext.jsCódigo
// CounterContext.js
import { createContext } from 'react';
export const CounterContext = createContext();

// App.js
const App = () => {
  const [count, setCount] = useState(0);
  return (
    <CounterContext.Provider value={{ count, setCount }}>
  <Controls />
  <Display />
    </CounterContext.Provider>
  );
};

// Controls.js
const Controls = () => {
  const { count, setCount } = useContext(CounterContext);
  return (
    <View style={styles.controls}>
  <Button title="-" onPress={() => setCount(count - 1)} />
  <Button title="Reset" onPress={() => setCount(0)} />
  <Button title="+" onPress={() => setCount(count + 1)} />
    </View>
  );
};

// Display.js
const Display = () => {
  const { count } = useContext(CounterContext);
  return (
    <View style={styles.display}>
  <Text style={styles.number}>{count}</Text>
    </View>
  );
};

// components/TodoInput.js
import React, { useState, useContext } from 'react';
import { View, TextInput, Button, StyleSheet } from 'react-native';
import { TodoContext, ThemeContext } from '../contexts';

const TodoInput = () => {
  const [text, setText] = useState('');
  const { todos, setTodos } = useContext(TodoContext);
  const { theme } = useContext(ThemeContext);

  const addTodo = () => {
    if (text.trim()) {
      setTodos([...todos, { id: Date.now(), text, completed: false }]);
      setText('');
    }
  };

  return (
    <View style={styles.container}>
      <TextInput
        style={[
          styles.input,
          { backgroundColor: theme === 'light' ? '#fff' : '#333',
            color: theme === 'light' ? '#000' : '#fff' }
        ]}
        value={text}
        onChangeText={setText}
        placeholder="Nova tarefa..."
        placeholderTextColor={theme === 'light' ? '#666' : '#999'}
      />
      <Button title="Adicionar" onPress={addTodo} />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    padding: 20,
    flexDirection: 'row',
  },
  input: {
    flex: 1,
    marginRight: 10,
    padding: 10,
    borderWidth: 1,
    borderRadius: 5,
  },
});

Passo 4: Componente TodoList

components/TodoList.jsCódigo
// components/TodoList.js
import React, { useContext } from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { TodoContext, ThemeContext } from '../contexts';

const TodoList = () => {
  const { todos, setTodos } = useContext(TodoContext);
  const { theme } = useContext(ThemeContext);

  const toggleTodo = (id) => {
    setTodos(todos.map(todo =>
      todo.id === id ? { ...todo, completed: !todo.completed } : todo
    ));
  };

  const deleteTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id));
  };

  return (
    <View style={styles.container}>
      {todos.map(todo => (
        <TouchableOpacity
          key={todo.id}
          style={[
            styles.todoItem,
            { backgroundColor: theme === 'light' ? '#f8f8f8' : '#333' }
          ]}
          onPress={() => toggleTodo(todo.id)}
          onLongPress={() => deleteTodo(todo.id)}
        >
          <Text style={[
            styles.todoText,
            { 
              color: theme === 'light' ? '#000' : '#fff',
              textDecorationLine: todo.completed ? 'line-through' : 'none'
            }
          ]}>
            {todo.text}
          </Text>
        </TouchableOpacity>
      ))}
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    padding: 20,
  },
  todoItem: {
    padding: 15,
    marginBottom: 10,
    borderRadius: 5,
    elevation: 2,
  },
  todoText: {
    fontSize: 16,
  },
});

// components/ThemeToggle.js
import React, { useContext } from 'react';
import { View, Switch, Text, StyleSheet } from 'react-native';
import { ThemeContext } from '../contexts';

const ThemeToggle = () => {
  const { theme, setTheme } = useContext(ThemeContext);

  return (
    <View style={styles.container}>
      <Text style={{ color: theme === 'light' ? '#000' : '#fff' }}>
        {theme === 'light' ? 'Tema claro' : 'Tema escuro'}
      </Text>
      <Switch
        value={theme === 'dark'}
        onValueChange={(value) => setTheme(value ? 'dark' : 'light')}
      />
    </View>
  );
};

const styles = StyleSheet.create({
  container: {
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'flex-end',
    padding: 20,
  },
});
