App.jsx · dentro de AppAtividade 2 / 02 de 07
function adicionar() {
  const titulo = texto.trim();
  if (!titulo) return;
  const nova = {
    id: String(proximoId.current),
    titulo,
    concluida: false,
  };
  proximoId.current += 1;
  setTarefas(atuais => [...atuais, nova]);
  setTexto('');
}

trim remove espaços nas pontas. O contador de IDs não volta ao remover tarefas.

Aula 16 / Resolução da atividade 2 / FlatList

Concluir, reabrir e remover

map atualiza a tarefa; filter exclui pelo identificador.

03 / 07
App.jsx · dentro de AppAtividade 2 / 03 de 07
function alternar(id) {
  setTarefas(atuais => atuais.map(tarefa =>
    tarefa.id === id
      ? { ...tarefa, concluida: !tarefa.concluida }
      : tarefa
  ));
}
function remover(id) {
  setTarefas(atuais => atuais.filter(tarefa => tarefa.id !== id));
}

As atualizações usam o estado anterior sem modificar os objetos existentes.

Refatore a atividade 2 para compartilhar as tarefas entre a lista e um resumo usando Context API.

const proximoId = useRef(4);

function adicionar(texto) {
  const titulo = texto.trim();

  const nova = {
    id: String(proximoId.current),

Aula 16 / Resolução da atividade 3 / useContext

Reaproveitar as outras ações

As funções de concluir e remover são as mesmas da atividade 2.

03 / 09
TarefasContext.jsx · dentro do ProviderAtividade 3 / 03 de 09
function alternar(id) {
  setTarefas(atuais => atuais.map(tarefa =>
    tarefa.id === id
      ? { ...tarefa, concluida: !tarefa.concluida }
      : tarefa
  ));
}
function remover(id) {
  setTarefas(atuais => atuais.filter(tarefa => tarefa.id !== id));
}

Mova as duas funções para o Provider: não é necessário reescrever a lógica.

Ler o contexto no resumo

O total de concluídas é calculado a partir das tarefas atuais.

import { useContext } from 'react';
import { TarefasContext } from './TarefasContext';

function ResumoTarefas() {
  const { tarefas } = useContext(TarefasContext);
  const concluidas = tarefas.filter(tarefa => tarefa.concluida).length;
  return (
    <Text style={{ padding: 12 }}>
      Concluídas: {concluidas} de {tarefas.length}
    </Text>
  );
}
