Tirar a foto

import { useRef, useState } from 'react';
import { Button, Image, Text, View } from 'react-native';
import { CameraView, useCameraPermissions } from 'expo-camera';

export default function App() {
  const [permission, requestPermission] = useCameraPermissions();
  const cameraRef = useRef(null);
  const [pronta, setPronta] = useState(false);
  const [foto, setFoto] = useState(null);

  async function fotografar() {
    if (!pronta || !cameraRef.current) return;
    try {
      const resultado = await cameraRef.current.takePictureAsync();
      setFoto(resultado.uri);
    } catch { alert('Falha ao fotografar.'); }
  }

  return (
    <CameraView
      style={{ width: '100%', height: 360 }}
      facing="back"
      ref={cameraRef}
      onCameraReady={() => setPronta(true)}
    />
  );
}

Obter a posição

import { useState } from 'react';
import { Button, Text, View } from 'react-native';
import * as Location from 'expo-location';

export default function App() {
  const [latitude, setLatitude] = useState('');
  const [longitude, setLongitude] = useState('');
  const [mensagem, setMensagem] = useState('');

  async function localizar() {
    setLatitude('');
    setLongitude('');
    setMensagem('');
    try {
      const permissao =
        await Location.requestForegroundPermissionsAsync();
      if (!permissao.granted) {
        setMensagem('Permissão negada.');
        return;
      }

      const ativa = await Location.hasServicesEnabledAsync();
      if (!ativa) {
        setMensagem('Ative a localização do aparelho.');
        return;
      }

      const resultado = await Location.getCurrentPositionAsync({
        accuracy: Location.Accuracy.Balanced,
      });
      setLatitude(resultado.coords.latitude);
      setLongitude(resultado.coords.longitude);
    } catch {
      setMensagem('Não foi possível localizar.');
    }
  }

  return (
    <View style={{ flex: 1, padding: 24, paddingTop: 64 }}>
      <Button title="Obter localização" onPress={localizar} />
      <Text>{mensagem}</Text>
    </View>
  );
}
Depois das duas verificações, obtenha uma posição e coloque cada coordenada no seu estado: setLatitude e setLongitude. Balanced é a precisão desejada, não uma garantia. A consulta pode demorar ou falhar; catch informa a falha. Em emulador ou simulador, forneça uma posição pelos controles da plataforma. O Auto-Animate preserva as linhas anteriores e destaca o trecho em construção. Fonte: Expo Location.

Acelerômetro: movimento em três eixos

xyz

Incline o celular e observe os eixos. As medidas são dadas em g (1 g aproximadamente 9,81 m/s²) e incluem a gravidade; parado não significa três zeros. O módulo costuma ficar perto de 1 g, com componentes diferentes conforme a posição do aparelho. O acelerômetro não fornece coordenadas geográficas. Fonte: Expo Accelerometer.

Receber as leituras

import { useEffect, useState } from 'react';

O retorno remove essa assinatura ao mudar lendo ou desmontar o componente. Assim, a limpeza já faz parte da leitura desde o início e repetidas chamadas setLendo(true) não duplicam listeners.
